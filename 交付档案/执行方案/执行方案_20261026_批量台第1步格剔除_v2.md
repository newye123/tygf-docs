## 1. 一句话目标与验收口径

在第 1 步点掉一格（某稿某门），**这一稿真的不再做那门**：顶栏份数、组卡份数、左栏份数、第 2 步页签、执行前确认检查、执行语言、回执与结算——七处跟着降；点「全选回来」七处逐字复原。

### 任务 1：纯函数（单点口径＋快照双写的内容层）

Files: Create `src/lib/workbench/slot-scope.ts`；Create `tests/unit/slot-scope.test.ts`

```ts
// src/lib/workbench/slot-scope.ts
import type { GroupLanguageSlot } from './group-binding'

/** 一稿的可交付份数＝该稿 ready 槽位语言 ∩ 该稿当前要做的语言集合。唯一口径，别处不许自己数。 */
export function countDeliverables(slots: GroupLanguageSlot[], languages: string[]): number {
  const doing = new Set(languages)
  return slots.filter((slot) => slot.state === 'ready' && doing.has(slot.lang)).length
}

/** 从一组语言里去掉一门；没有该门时返回同一引用（不制造无意义的数组拷贝）。 */
export function withoutLanguage(languages: string[], language: string): string[] {
  if (!languages.includes(language)) return languages
  return languages.filter((item) => item !== language)
}

/** 「全选回来」的目标＝该稿所有 ready 槽位语言（数据派生，不靠记点击）。 */
export function readySlotLanguages(slots: GroupLanguageSlot[]): string[] {
  return slots.filter((slot) => slot.state === 'ready').map((slot) => slot.lang)
}

export function withLanguages(languages: string[], add: string[]): string[] {
  const out = [...languages]
  for (const language of add) if (!out.includes(language)) out.push(language)
  return out
}
```

房子写法说明：v1 用 `const { [key]: _drop, ...rest }`（N-1，`src/` 0 先例）已全部去掉；删键用 spread+delete（现例 `workbench-view.tsx:2238-2239`）。
